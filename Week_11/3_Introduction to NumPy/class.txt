import numpy as np
import time


# ============================================================
# 1. Python Lists vs NumPy Arrays
# ============================================================

# Python Lists:
# - Can store different data types
# - Can grow or shrink dynamically
# - Do not directly support element-wise mathematical operations
# - Multidimensional data can be difficult to manage
# - Less efficient for large numerical computations
#
# NumPy Arrays:
# - Fixed size after creation
# - Usually contain elements of the same data type
# - Support element-wise operations
# - Efficient for large numerical data
# - Easy to work with multidimensional data
# - Better memory and computational efficiency
# - Widely used in Data Science, Machine Learning, and Scientific Computing

# ============================================================
# 34. Performance Comparison
# ============================================================

n = 10_000_000


# -----------------------------
# Python List
# -----------------------------

python_list = range(n)

start = time.time()

result_list = [i ** 2 for i in python_list]

end = time.time()

time_taken_list = end - start

print("\nPython List Time:")
print(time_taken_list)


# -----------------------------
# NumPy Array
# -----------------------------

numpy_array = np.arange(n)

start = time.time()

result_array = numpy_array ** 2

end = time.time()

time_taken_array = end - start

print("\nNumPy Array Time:")
print(time_taken_array)


# ============================================================
# 35. Performance Result
# ============================================================

print("\nPerformance Comparison:")
print("Python List Time :", time_taken_list)
print("NumPy Array Time :", time_taken_array)

if time_taken_array < time_taken_list:
    print("NumPy is faster for this operation.")
else:
    print("Python List was faster for this particular run.")
